\section{Kärtner Kasnudeln}
\begin{zutaten}
 500g&Schär Farine\\
 70g&Wasser\\
 40g&Olivenöl\\
 3&Eier\\
 500ml&Milch-Wasser Gemisch\\
 250ml&Hirse\\
 1EL&Salz\\
 500g&Bauerntopfen\\
 &Minze\\
 &Kerbel\\
 &Majv\\
 &Butter\\
\end{zutaten}
\begin{enumerate}
 \item Mehl, Wasser, Olivenöl und Eier zu einem Teig kneten.
 \item Hirse Waschen.
 \item Milch-Wasser Gemisch mit Hirse kochen.
 \item Auskühlen lassen.
 \item Topfen und Gewürze untermischen.
 \item Teig 1 - 1,5mm dick ausrollen.
 \item Runde Stücke aus dem Teig herausstechen.
 \item Jeweils 2 Stücke Teig mit Füllung füllen und den Rand krendeln.
 \item In kochendem, gesalzenen Wasser solange kochen bis sie schwimmen.
 \item In Butter goldbraun braten.
\end{enumerate}
\Kapitel{Brot und Brötchen}{pics/brot}
\section{Sauerteig}
\begin{zutaten}
260g&Mehl\\
260g&Wasser\\
\end{zutaten}
\begin{enumerate}
 \item Tag 1: 30g Mehl und 30g Wasser in einem verschließbaren Gefäß vermengen und offen stehen lassen.
 \item Tag 2-4: 30g Mehl und 30g Wasser zum Ansatz hinzufügen und Gefäß offen stehen lassen.
 \item Tag 5: 30g vom Ansatz herauswiegen und mit 70g Mehl und 70g Wasser vermengen und offen stehen lassen. Den Rest entsorgen.
 \item Tag 6: 70g Mehl und 70g Wasser zum Ansatz hinzufügen und Gefäß offen stehen lassen.
 \item Tag 7: Erstes Brot kann gemacht werden. Bei den ersten Versuchen etwas Trockenhefe dazugeben, da der Sauerteig noch nicht genug Kraft entwickelt hat.
\end{enumerate}
\section{Sauerteigbrot}
\begin{zutaten}
140g&Mehl\\
140g&Wasser\\
30g&Sauerteigansatz\\
500g&Mehl\\
450ml&Wasser\\
10g&Salz\\
4EL&Olivenöl\\
\end{zutaten}
\ofen{ul}{220-250}{50min}
\begin{enumerate}
 \item Am Vortag Sauerteigansatz mit 140g Mehl und 140g Wasser vermengen und verschlossen stehen lassen.
 \item Vom Sauerteigansatz 30g herauswiegen und für das nächste Mal verschlossen in den Kühlschrank stellen.
 \item Den Sauerteigansatz im Wasser auflösen.
 \item Mit Mehl, Salz und Olivenöl vermengen und für 10-15 Minuten kneten.
 \item Gärkörbchen oder Kastenform ausmehlen und den Teig über Nacht (min. 8 Stunden) zugedeckt an einem warmen Ort gehen lassen.
 \item Ofen auf 250°C Umluft mit einer Schale Wasser vorheizen.
 \item Brot aus dem Gärkörbchen stürzen und für 50 Minuten backen. Nach 10 Minuten Temperatur auf 220°C reduzieren.
\end{enumerate}

\section{English Muffins}
\begin{zutaten}
 200g&Mehl\\
 7g&Zucker\\
 5g&Salz\\
 ½ Pkg&Backpulver\\
 250ml&warme Milch\\
 25g&geschmolzene Butter\\
 &Polenta\\
\end{zutaten}
\begin{enumerate}
 \item Mehl, Zucker, Salz und Backpulver in einer Schüssel vermengen.
 \item Milch und Butter einrühren bis ein Teig entsteht.
 \item Teig für ca. 2 Stunden an einem warmen Ort gehen lassen.
 \item Auf einer leicht bemehlten Fläche ungefähr 1cm dick ausrollen und mit einem runden Keksausstecher oder einem Glas ausstechen.
 \item Muffins mit Polenta beidseitig bedecken.
 \item Nochmals bedeckt 30 Minuten gehen lassen.
 \item In einer geölten Pfanne ca. 5-7 Minuten pro Seite goldbraun backen.
\end{enumerate}

\Kapitel{Kuchen und Torten}{pics/torte}
\section{Schwarzwälder Kirschtorte}
\begin{zutaten}
5&Eier\\
1 Prise&Salz\\
200g&Zucker\\
2 Pkg&Vanillezucker\\
250g&Mehl\\
1 Pkg&Backpulver\\
3 EL&Kakaopulver\\
1 EL&Rum\\
800g&Kirschen\\
400g&Gelierzucker 2:1\\
500ml&Schlagobers\\
2 Pkg&Sahnesteif\\
1 Schuss&Zitronensaft\\
200g&Schokoraspeln\\
\end{zutaten}
\ofen{ul}{175}{25-30min}
\begin{enumerate}
 \item Kirschen entkernen und mit dem Gelierzucker zum Kochen bringen, anschließend köcheln lassen.
 \item Ofen auf 190°C (Umluft 175°C) vorheizen. Eier, Mehl, Zucker, Vanillezucker, Backpulver, Kakao und Rum in einer Rührschüssel glatt rühren.
 \item Teig in eine Springform geben und für 25-30 Minuten backen.
 \item Den fertigen Boden in 3 Teile schneiden.
 \item Boden und Kirschmasse kühlen lassen.
 \item Schlagobers mit Sahnesteif steif schlagen und Zitronensaft zugeben.
 \item Den ersten Boden auf eine Platte legen und den Springformrand oder einen Toretenring darum herum stellen. Darauf die Kirschmasse gleichmäßig verteilen.
 \item Den zweiten Boden auflegen und leicht andrücken. Ein Drittel des Schlagobers gleichmäßig darauf streichen.
 \item Den dritten Boden auflegen und leicht andrücken. Den restlichen Schlagobers gleichmäßig auf und um die Torte streichen. Mit Schokoraspeln garnieren.
 \item 16 Tupfen Schlagobers gleichmäßig am Rand der oberfläche verteilen und darauf entkernte Kirschen platzieren. Torte vor dem servieren 3-4 Stunden kaltstellen.
\end{enumerate}
\section{Blechkuchen \textsc{\textmd{\small{by Auriel Orion Ené}}}}
\begin{zutaten}
 325g&Mehl\\
 150g&Zucker\\
 175ml&Milcgh\\
 4&Eier\\
 160g&Butter/Rapsöl\\
 1 Prise&Salz\\
 1 Pkg&Backpulver\\
 1 Pkg&Vanillezucker\\
 200-400g&Obst\\
\end{zutaten}
\ofen{ou}{180}{25-30min}
\begin{enumerate}
 \item Ofen auf 180°C Ober-/Unterhitze vorheizen.
 \item Butter, Zucker, Vanillezucker und Eier schaumig rühren.
 \item Mehl mit Backpulver und Salz mischen und Esslöffelweise unterrühren.
 \item Die Milch schluckweise dazugeben und verrühren.
 \item Tiefes Backblech gut einfetten und den Teig gleichmäßig darauf verteilen.
 \item Obst auf dem Teig verteilen und leicht in den Teig drücken.
 \item 25-30min backen.
 \item Kuchen auskühlen lassen und mit Staubzucker bestreuen.
\end{enumerate}
\section{Karottenkuchen \textsc{\textmd{\small{by Auriel Orion Ené}}}}
\begin{zutaten}
 200g&Mehl\\
 250g&Zucker\\
 2TL&Backpulver\\
 1 Prise&Salz\\
 200g&Walnüsse gerieben\\
 350g&Karotten\\
 4&Eier\\
 100ml&Öl\\
 1&Zitrone\\
 200g&Staubzucker\\
 &Zitronensaft\\
\end{zutaten}
\ofen{ou}{180}{50min}
\begin{enumerate}
 \item Ofen auf 180°C Ober-/Unterhitze (Umluft 160°C) vorheizen und Form einfetten.
 \item Karotten schälen und fein reiben.
 \item Karotten mit Salz, Mehl, Walnuss, Backpulver und geriebener Zitronenschale mischen.
 \item Eier trennen, Eiklar steif schlagen und zur Seite stellen.
 \item Eigelb mit Zucker, Öl und etwas Zitronensaft verrühren.
 \item Die Karotten-Mehl Masse zur Eiermasse gebe und kurz durchrühren.
 \item Eiklar unterheben und Teig in die Form füllen.
 \item Kuchen für ca. 50 Minuten backen.
 \item Staubzucker mit Zitronensaft mischen.
 \item Kuchen auskühlen lassen und mit der Staubzucker-Zitronensaft Mischung übergießen.
\end{enumerate}
\section{Himbeertiramisu \textsc{\textmd{\small{by Auriel Orion Ené}}}}
\begin{zutaten}
 &Biskotten\\
 500g&Mascarpone\\
 4&Eier\\
 250ml&Schlagobers\\
 100g&Staubzucker\\
 500-600g&Himbeeren\\
 &Zitronensaft\\
 &Butter
\end{zutaten}
\begin{enumerate}
 \item Himbeeren mit etwas Staubzucker, Zitronensaft und Butter in einem Topf geben und kochen bis eine Soße entsteht.
 \item Eier trennen und Eiklar schaumig schlagen.
 \item Schlagobers schaumig schlagen.
 \item Eidotter und Staubzucker mischen, Mascarpone und Schlag unterrühren.
 \item Eiklar unterheben.
 \item Tiramisu in einer Form aufschichten. Zuerst Biskotten, darauf Soße und dann Mascarpone-Mischung. Solange wiederholen bis die Form gefüllt ist.
 \item Tiramisu mit Spekulatiusbrösel oder Kakao bestreuen.
 \item Für mindestens 4h kalt stellen.
\end{enumerate}
\section{Nutellaschnitte}
\begin{zutaten}
 7&Eier\\
 300g&Staubzucker\\
 280g&Mehl\\
 125ml&Öl\\
 125ml&Wasser\\
 ½ Pkg&Backpulver\\
 4 EL&Kakao\\
 100g&Zucker\\
 250g&Qimiq\\
 250g&Qimiq Vanille\\
 100g&Nutella\\
 250ml&Schlagobers\\
 &Schokostreusel\\
\end{zutaten}
\ofen{ou}{200}{15min}
\begin{enumerate}
 \item Ofen auf 200°C Ober-/Unterhitze vorheizen.
 \item Tiefes Backblech einfetten oder mit Backpapier auslegen.
 \item Eier und Staubzucker schaumig rühren.
 \item Mehl, Backpulver, Kakao, Öl und Wasser unterrühren.
 \item Für ca. 15 Minuten backen.
 \item Zucker, Quimiq, Nutella und Schlagobers zu einer Creme verrühren.
 \item Creme auf dem Kuchen verstreichen und mit Schokostreuseln bestreuen.
 \item Kalt stellen.
\end{enumerate}
